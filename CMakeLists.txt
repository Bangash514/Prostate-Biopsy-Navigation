cmake_minimum_required(VERSION 3.10)
project(ProstateBiopsyNavigation CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED)
add_executable(navigation_demo cpp/navigation_demo.cpp)
target_include_directories(navigation_demo PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(navigation_demo Eigen3::Eigen ${OpenCV_LIBS})
target_compile_options(navigation_demo PRIVATE -Wall -Wextra)
